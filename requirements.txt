# Python 3.9 and 3.15+ are outside this range: current NumPy, SciPy, pandas,
# and pyarrow releases do not publish wheels for them, and installing from
# source fails with "No usable build for your environment".
#
# Prefer:
#   python -m pip install --only-binary=numpy,scipy,pandas,pyarrow,pillow -r requirements.txt

numpy>=1.26.4,<2.3 ; python_version == "3.10"
scipy>=1.11.4,<1.16 ; python_version == "3.10"
pandas>=2.2.2,<2.4 ; python_version == "3.10"

pandas>=2.2.2,<3.1 ; python_version == "3.11"

numpy>=1.26.4,<2.6 ; python_version >= "3.12" and python_version < "3.15"
scipy>=1.11.4,<1.19 ; python_version >= "3.12" and python_version < "3.15"
pandas>=2.2.2,<3.1 ; python_version >= "3.12" and python_version < "3.15"

# Streamlit imports pyarrow. Version 25.0.0 is excluded upstream; 25.0.1
